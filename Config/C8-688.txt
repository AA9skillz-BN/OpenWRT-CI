# =====================================================================
# NRadio / 鲲鹏无限 C8-688（nradio,c8-688 / MT7981B + MT7531 + eMMC）
# 单机型配置片段：只编 C8-688，不编整个 filogic 家族
#
# 配套：
#   .github/workflows/C8-688.yml     薄壳（uses: WRT-CORE.yml + 产物闸门）
#   Scripts/C8-688/private.sh        注入 DTS / 设备定义 / 02_network / 覆盖层
#   Files/C8-688/                    风扇曲线、首启
#   docs/C8-688.md
#
# 注意：WRT-CORE 用 `CONFIG_TARGET_<X>=y` 推导平台，前两行不能动；
#       板级包**必须在这里显式选**，只写 filogic.mk 的 DEVICE_PACKAGES 不会进镜像
#       （2026-10-26 实测：闸门抓到镜像缺 kmod-mt7915e）。
# =====================================================================

# 平台 / 机型
CONFIG_TARGET_mediatek=y
CONFIG_TARGET_mediatek_filogic=y
CONFIG_TARGET_ALL_PROFILES=n

# ---------------------------------------------------------------------

# ---------------------------------------------------------------------

# 2.5G PHY：GPY211 x2（实机在 mdio 5 / mdio 15）

# USB：MT5700M 模组挂在主 SoC USB 上（3466:3301，ttyUSB0~3）
CONFIG_PACKAGE_kmod-usb-acm=y
CONFIG_PACKAGE_kmod-usb-net-qmi-wwan=y
